cmake_minimum_required(VERSION 3.20)
project(lagotracks_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(LT_WITH_JUCE "Saída de áudio pelo JUCE" ON)
set(LT_PYBIND11_DIR "" CACHE PATH "pybind11 já baixado (opcional)")
set(LT_JUCE_DIR "" CACHE PATH "JUCE já baixado (opcional)")

include(FetchContent)
set(PYBIND11_FINDPYTHON ON)
if(LT_PYBIND11_DIR)
  add_subdirectory(${LT_PYBIND11_DIR} pybind11 EXCLUDE_FROM_ALL)
else()
  FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git GIT_TAG v2.13.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(pybind11)
endif()

set(LT_SOURCES src/core.cpp src/bindings.cpp)
if(LT_WITH_JUCE)
  if(LT_JUCE_DIR)
    add_subdirectory(${LT_JUCE_DIR} juce EXCLUDE_FROM_ALL)
  else()
    FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(juce)
  endif()
  list(APPEND LT_SOURCES src/device_juce.cpp)
else()
  list(APPEND LT_SOURCES src/device_stub.cpp)
endif()

pybind11_add_module(_native ${LT_SOURCES})
target_include_directories(_native PRIVATE src)
if(MSVC)
  target_compile_options(_native PRIVATE /O2 /W3 /EHsc /fp:precise)
else()
  target_compile_options(_native PRIVATE -O2 -Wall -Wextra)
endif()
if(LT_WITH_JUCE)
  target_compile_definitions(_native PRIVATE LT_WITH_JUCE=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_MODAL_LOOPS_PERMITTED=0 JUCE_STRICT_REFCOUNTEDPOINTER=1 JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_REPORT_APP_USAGE=0 JUCE_USE_DIRECTWRITE=0)
  target_link_libraries(_native PRIVATE juce::juce_audio_devices juce::juce_events juce::juce_core
                                        juce::juce_audio_basics juce::juce_recommended_config_flags)
endif()
set_target_properties(_native PROPERTIES OUTPUT_NAME "_native")
